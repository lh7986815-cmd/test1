    [Transaction(TransactionMode.Manual)]
    public class Test : IExternalCommand
    {
        public Result Execute(ExternalCommandData commandData, ref string message, ElementSet elements)
        {
            UIDocument uidoc = commandData.Application.ActiveUIDocument;
            Document doc = uidoc.Document;

            try
            {
                Reference r = uidoc.Selection.PickObject(ObjectType.Element, "请选择要设置立柱的栏杆");
                Railing railing = doc.GetElement(r) as Railing;

                if (railing != null)
                {
                    PostConfigWindow win = new PostConfigWindow(doc, railing);
                    bool? result = win.ShowDialog();

                    if (result == true)
                    {
                        uidoc.RefreshActiveView();
                    }
                }
                return Result.Succeeded;
            }
            catch (Autodesk.Revit.Exceptions.OperationCanceledException)
            {
                return Result.Cancelled;
            }
            catch (Exception ex)
            {
                Autodesk.Revit.UI.TaskDialog.Show("运行报错", ex.Message);
                return Result.Failed;
            }
        }
    }

        private FamilySymbol GetOrCreateSym(Family baseFamily, FamilySymbol baseSym, string prefix, string wText, string hText, string tText)
        {
            string symName = $"{prefix}_{wText}x{hText}x{tText}mm";
            FamilySymbol sym = null;

            // 1. 安全地遍历查找，避免将 null 传给 GetElement
            foreach (ElementId id in baseFamily.GetFamilySymbolIds())
            {
                FamilySymbol existingSym = _doc.GetElement(id) as FamilySymbol;
                if (existingSym != null && existingSym.Name == symName)
                {
                    sym = existingSym;
                    break; // 找到了对应的尺寸类型，直接跳出循环
                }
            }

            // 2. 如果没找到 (sym 依然是 null)，说明是新尺寸，执行复制新建
            if (sym == null)
            {
                sym = baseSym.Duplicate(symName) as FamilySymbol;

                Parameter pW = sym.LookupParameter("宽度");
                Parameter pH = sym.LookupParameter("高度");
                Parameter pT = sym.LookupParameter("厚度");

                if (pW != null && !pW.IsReadOnly) pW.Set(double.Parse(wText) / 304.8);
                if (pH != null && !pH.IsReadOnly) pH.Set(double.Parse(hText) / 304.8);
                if (pT != null && !pT.IsReadOnly) pT.Set(double.Parse(tText) / 304.8);
            }

            // 确保激活状态
            if (!sym.IsActive) sym.Activate();
            return sym;
        }

        // --- 核心写入逻辑 ---
        private void BtnApply_Click(object sender, System.Windows.RoutedEventArgs e)
        {
            try
            {
                using (Transaction t = new Transaction(_doc, "一键设置所有立柱"))
                {
                    t.Start();

                    string rfaPath = @"D:\test\test1\test1\中间立柱.rfa";
                    Family fam = new FilteredElementCollector(_doc).OfClass(typeof(Family)).Cast<Family>().FirstOrDefault(f => f.Name == "中间立柱");

                    if (fam == null)
                    {
                        if (!System.IO.File.Exists(rfaPath)) { System.Windows.MessageBox.Show("找不到立柱族文件: " + rfaPath); return; }
                        _doc.LoadFamily(rfaPath, out fam);
                    }

                    FamilySymbol baseSym = _doc.GetElement(fam.GetFamilySymbolIds().First()) as FamilySymbol;
                    if (!baseSym.IsActive) baseSym.Activate();

                    FamilySymbol startSym = GetOrCreateSym(fam, baseSym, "起点", txtStartW.Text, txtStartH.Text, txtStartT.Text);
                    FamilySymbol endSym = GetOrCreateSym(fam, baseSym, "终点", txtEndW.Text, txtEndH.Text, txtEndT.Text);
                    FamilySymbol cornerSym = null;
                    if (chkCorner.IsChecked == true) cornerSym = GetOrCreateSym(fam, baseSym, "转角", txtCornerW.Text, txtCornerH.Text, txtCornerT.Text);

                    // ... 前面的获取和复制族类型的代码保持不变 ...

                    // 4. 提取并修改立柱系统结构 (它是实时代理，拿出来直接改！)
                    RailingType rType = _doc.GetElement(_railing.GetTypeId()) as RailingType;
                    BalusterPlacement bp = rType.BalusterPlacement;
                    PostPattern postPat = bp.PostPattern;

                    if (postPat.StartPost != null) postPat.StartPost.BalusterFamilyId = startSym.Id;
                    if (postPat.EndPost != null) postPat.EndPost.BalusterFamilyId = endSym.Id;

                    if (chkCorner.IsChecked == true)
                    {
                        if (postPat.CornerPost != null) postPat.CornerPost.BalusterFamilyId = cornerSym.Id;
                    }
                    else
                    {
                        // 如果不勾选，则设为 <无>
                        if (postPat.CornerPost != null) postPat.CornerPost.BalusterFamilyId = ElementId.InvalidElementId;
                    }

                    // 🚨 终极奥义：把下面这行彻底删掉！不需要 Set，也不需要 get_Parameter 赋值！
                    // rType.SetBalusterPlacement(bp);  <--- 就是它惹的祸，删！

                    // 直接强制重算模型并提交，大功告成
                    _doc.Regenerate();
                    t.Commit();
                }

                this.DialogResult = true;
                this.Close();
            }
            catch (Exception ex)
            {
                System.Windows.MessageBox.Show(ex.Message + "\n" + ex.StackTrace, "发生错误");
            }
        }
